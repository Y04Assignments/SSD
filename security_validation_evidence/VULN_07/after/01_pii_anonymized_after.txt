export const getActiveSellRequests = async (req, res) => {
  try {
    // NO .populate('resident') - resident data is NOT fetched at all
    const rawRequests = await SellRequest.find({ status: 'Pending' })
      .sort({ createdAt: -1 })
      .select('energyAmount location comment status createdAt')  // Select ONLY safe fields
      .lean();

    const requests = rawRequests.map(request => {
      const coords = request.location?.coordinates;
      
      // FUZZ coordinates to 2 decimal places (~1.1km area) 
      // Prevents exact residential tracking
      const fuzzedLocation =
        coords && Array.isArray(coords) && coords.length === 2
          ? {
              type: 'Point',
              coordinates: [
                Math.round(Number(coords[0]) * 100) / 100,  // 80.123456 -> 80.12
                Math.round(Number(coords[1]) * 100) / 100,  // 7.654321 -> 7.65
              ],
            }
          : request.location;

      // Generate anonymous username - never expose real name or email
      const sellerSuffix = request._id ? request._id.toString().slice(-4) : 'Resident';

      return {
        _id: request._id,
        username: `Solar Seller #${sellerSuffix}`,  // Anonymous identifier
        energyAmount: request.energyAmount,
        location: fuzzedLocation,    // Fuzzed, not exact
        comment: request.comment,
        status: request.status,
        createdAt: request.createdAt,
        // resident: ABSENT - not included in response
        // email: ABSENT - not included in response  
        // name: ABSENT - not included in response
      };
    });

    res.status(200).json({ message: 'Active sell requests retrieved successfully', requests });
  }
};
