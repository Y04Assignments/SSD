/**
 * Escapes special regular expression metacharacters to treat user input as literal text.
 * @param {string} str
 * @returns {string}
 */
export const escapeRegex = str => {
  if (typeof str !== 'string') return '';
  // Escapes: . * + ? ^ $ { } ( ) | [ ] \
  return str.replace(/[.*+?^${}()|[\]\\]/g, '\\$&');
};
