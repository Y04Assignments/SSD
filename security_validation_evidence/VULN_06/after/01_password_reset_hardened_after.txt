// Generate cryptographically secure 6-digit reset code
const resetCode = crypto.randomInt(100000, 1000000).toString();
// crypto.randomInt uses CSPRNG (cryptographically secure) - NOT Math.random()

// Hash reset code with SHA-256 before storing in database
const hashedResetCode = crypto.createHash('sha256').update(resetCode).digest('hex');

// Save HASHED token (never plaintext) + rate limit fields
user.passwordResetToken = hashedResetCode;    // SHA-256 hash, not plaintext
user.passwordResetExpires = new Date(Date.now() + 10 * 60 * 1000); // 10 minutes
user.passwordResetAttempts = 0;               // Reset attempt counter

=== FIXED SOURCE CODE - userController.js: resetPassword ===

const MAX_ATTEMPTS = 5;

// Check if account is locked (15-minute lockout)
if (user.passwordResetLockUntil && user.passwordResetLockUntil.getTime() > Date.now()) {
  return res.status(429).json({   // HTTP 429 Too Many Requests
    message: 'Too many failed attempts. Please request a new password reset.',
  });
}

// Check attempt counter BEFORE comparing
if (user.passwordResetAttempts >= MAX_ATTEMPTS) {
  user.passwordResetToken = null;
  user.passwordResetLockUntil = new Date(Date.now() + 15 * 60 * 1000); // 15-min lockout
  await user.save();
  return res.status(429).json({ ... });
}

// Hash provided code and compare using CONSTANT-TIME comparison (timing-safe)
const hashedInput = crypto.createHash('sha256').update(String(resetCode).trim()).digest('hex');
const storedBuf = Buffer.from(user.passwordResetToken || '');
const inputBuf = Buffer.from(hashedInput);

const isMatch =
  storedBuf.length === inputBuf.length &&
  crypto.timingSafeEqual(storedBuf, inputBuf);   // Timing-safe comparison

if (!isMatch) {
  user.passwordResetAttempts = (user.passwordResetAttempts || 0) + 1;
  if (user.passwordResetAttempts >= MAX_ATTEMPTS) {
    // Lock account on 5th failure
    user.passwordResetToken = null;
    user.passwordResetLockUntil = new Date(Date.now() + 15 * 60 * 1000);
    await user.save();
    return res.status(429).json({ message: 'Too many failed attempts...' });
  }
  await user.save();
  return res.status(400).json({ message: 'Invalid or expired reset code' });
}

// On success: invalidate token immediately (single-use enforcement)
user.password = newPassword;
user.passwordResetToken = null;       // Token invalidated after use
user.passwordResetExpires = null;
user.passwordResetAttempts = 0;
user.passwordResetLockUntil = null;
