// Short-lived single-use authorization code store (TTL: 60s)
export const oauthAuthCodes = new Map();

// Google OAuth callback handler
router.get('/google/callback', passport.authenticate('google', { ... }), async (req, res) => {
  const user = req.user;
  const userData = { id: user._id.toString(), email: user.email, name: user.name, role: user.role };
  
  const token = jwt.sign(userData, process.env.JWT_SECRET, { expiresIn: '30d' });

  // Generate single-use, high-entropy authorization code (NEVER the token itself)
  const authCode = crypto.randomBytes(32).toString('hex');   // 256-bit random code
  oauthAuthCodes.set(authCode, {
    token,
    user: userData,
    expiresAt: Date.now() + 60000, // 60-second TTL
  });

  // HttpOnly cookie for session security (inaccessible to JavaScript)
  res.cookie('token', token, {
    httpOnly: true,    // Cannot be read by JavaScript
    secure: process.env.NODE_ENV === 'production',  // HTTPS only in production
    sameSite: 'lax',   // CSRF protection
    maxAge: 30 * 24 * 60 * 60 * 1000,
  });

  // Secure redirect: ONLY the one-time authorization code in URL
  // NEVER the JWT token or user data!
  res.redirect(`${frontendUrl}/oauth/callback?code=${authCode}`);
  // The URL contains: ?code=7f3a9b2e... (one-time, 60s TTL, random, opaque)
  // NOT: ?token=eyJhbGci... (the JWT is NEVER in the URL)
});

// Token exchange endpoint - converts one-time code to JWT
router.post('/exchange', async (req, res) => {
  const { code } = req.body;
  const entry = oauthAuthCodes.get(code);
  
  if (!entry) {
    return res.status(400).json({ message: 'Invalid or expired authorization code' });
  }
  
  // IMMEDIATELY delete code - single-use enforcement
  oauthAuthCodes.delete(code);   // Code can NEVER be reused
  
  if (entry.expiresAt <= Date.now()) {
    return res.status(400).json({ message: 'Authorization code has expired' });
  }
  
  return res.status(200).json({ success: true, token: entry.token, user: entry.user });
});

=== SECURITY ARCHITECTURE SUMMARY ===

URL: /oauth/callback?code=7f3a9b2e4d...   (opaque 32-byte random code)
  vs. BEFORE: /oauth/callback?token=eyJhbGci... (full JWT in URL)

The authorization code:
  ✓ Is 256 bits of cryptographic randomness (not guessable)
  ✓ Has a 60-second expiry
  ✓ Is single-use (deleted immediately on redemption)
  ✓ Cannot be replayed even if intercepted
  ✓ Does not expose any user data directly
